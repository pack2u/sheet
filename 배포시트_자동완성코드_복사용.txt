// [용도]
// - 배포 시트의 onEdit 자동완성 로직을 수동으로 복구/재설치할 때 쓰는 복사용 원본입니다.
// [언제 사용]
// - 자동 주입(createViewerNoticeScript_) 실패, 권한/API 오류, 개별 시트만 긴급 복구가 필요할 때.
// [적용 방법]
// - 대상 배포시트의 Apps Script 편집기에 붙여넣고 저장한 뒤, onEdit 트리거 동작을 테스트하세요.
// [주의]
// - 본 파일은 실행 파일이 아니라 운영 백업 스니펫입니다.
// - A열(거래처명)은 ARRAYFORMULA가 자동 처리.
// - D열(품목명), L열(단가)은 C열 입력 시 onEdit가 뷰어 탭에서 조회하여 "값"을 직접 기록 (수식 없음).
// ── [Pack2U] 발주 및 송장조회 탭 — 값 기반 자동완성 (2026-06-17 전환) ──
function onEdit(e) {
  function cleanCode_(str) {
    if (!str) return '';
    return String(str)
      .replace(/\u00a0/g, '')
      .replace(/[\u200b-\u200d\ufeff]/g, '')
      .replace(/[\x00-\x1F\x7F-\x9F]/g, '')
      .replace(/\s/g, '')
      .toUpperCase();
  }
  try {
    if (!e || !e.range) return;
    var sheet = e.range.getSheet();
    var sheetName = sheet.getName();

    // ── [보강] 단가조회 탭 서식 오염 방지 및 조건부서식 재구축 ──
    if (sheetName === '단가조회' || sheetName === '팩투유 단가조회') {
      var r = e.range;
      var row = r.getRow();
      var numRows = r.getNumRows();
      sheet.getRange(row, 1, numRows, 10).setBackground(null);
      try {
        sheet.clearConditionalFormatRules();
        var vRange = sheet.getRange("A3:J5000");
        var rules = [];
        rules.push(
          SpreadsheetApp.newConditionalFormatRule()
            .whenFormulaSatisfied('=ISNUMBER(SEARCH("품절", $A3))')
            .setBackground("#f4cccc")
            .setRanges([vRange])
            .build()
        );
        rules.push(
          SpreadsheetApp.newConditionalFormatRule()
            .whenFormulaSatisfied('=ISNUMBER(SEARCH("단종", $A3))')
            .setBackground("#d9d9d9")
            .setRanges([vRange])
            .build()
        );
        rules.push(
          SpreadsheetApp.newConditionalFormatRule()
            .whenFormulaSatisfied('=ISNUMBER(SEARCH("재고까지만", $A3))')
            .setBackground("#ffe599")
            .setRanges([vRange])
            .build()
        );
        sheet.setConditionalFormatRules(rules);
      } catch(errDesign) {}
      return;
    }

    if (sheetName !== '발주 및 송장조회') return;

    var r = e.range;
    var row = r.getRow();
    var numRows = r.getNumRows();
    var startCol = r.getColumn();
    var numCols = r.getNumColumns();
    if (row < 2 || numRows <= 0) return;
    if (numRows > 500) return;

    // ★ 2026-06-22: 자동입력 열 보호: A(1), D(4), L(12), M(13) — 수동 입력 시 즉시 차단 + 뷰어탭 값 복구
    var _protCols = [1, 4, 12, 13];
    var _hadProt = false;
    for (var pi = 0; pi < _protCols.length; pi++) {
      var pc = _protCols[pi];
      if (pc >= startCol && pc < startCol + numCols) {
        sheet.getRange(row, pc, numRows, 1).clearContent();
        _hadProt = true;
      }
    }

    // C열(코드), D열(품목명), L열(단가) 수정 감지
    var hasC = (startCol <= 3 && startCol + numCols > 3);
    var hasD = (startCol <= 4 && startCol + numCols > 4);
    var hasL = (startCol <= 12 && startCol + numCols > 12);
    var hasEH = (startCol <= 8 && startCol + numCols > 5);
    if (!hasC && !hasD && !hasL && !hasEH) return;

    // [안전성] 뷰어(단가조회) 탭 동적 탐색
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var viewerTab = ss.getSheetByName('단가조회');
    if (!viewerTab) {
      var allTabs = ss.getSheets();
      for (var t = 0; t < allTabs.length; t++) {
        var tn = allTabs[t].getName();
        if (tn.indexOf('마감') !== -1 || tn.indexOf('발주') !== -1 || tn.indexOf('설정') !== -1 || tn.indexOf('검색') !== -1) continue;
        if (tn.indexOf('단가조회') !== -1 || tn.indexOf('뷰어') !== -1 || tn.indexOf('팩투유') !== -1 || tn.indexOf('단가') !== -1) {
          viewerTab = allTabs[t]; break;
        }
      }
    }
    if (!viewerTab) viewerTab = ss.getSheets()[0];
    var vLast = viewerTab.getLastRow();
    if (vLast < 4) return;
    // Row1=공지, Row2=헤더, Row3=숨긴 ARRAYFORMULA행 → 실제 코드는 Row4부터
    var vData = viewerTab.getRange(4, 1, vLast - 3, 7).getValues(); // A~G열

    // B열(주문일자): C 입력이 있는데 비어있으면 오늘 날짜(yyyyMMdd) 자동입력
    try {
      var bcdData = sheet.getRange(row, 2, numRows, 2).getValues(); // B:C
      var todayYmd = Utilities.formatDate(new Date(), 'Asia/Seoul', 'yyyyMMdd');
      var bVals = [];
      var bChanged = false;
      for (var bi = 0; bi < bcdData.length; bi++) {
        var curB = bcdData[bi][0];
        var curC = String(bcdData[bi][1] || '').trim();
        if (curC && !String(curB || '').trim()) {
          curB = todayYmd;
          bChanged = true;
        }
        bVals.push([curB]);
      }
      if (bChanged) sheet.getRange(row, 2, numRows, 1).setValues(bVals);
    } catch (eDateFill) {}

    // ★ C열 코드 입력 또는 D/L열 직접 편집 시: 빴어 탭에서 조회하여 D/L 값 기록 + J열 경고
    if (hasC || hasD || hasL) {
      for (var i = 0; i < numRows; i++) {
        var inputCode = cleanCode_(sheet.getRange(row + i, 3).getValue());

        // 코드가 비어있으면 D/L도 클리어 (C열 편집 시만)
        if (!inputCode) {
          if (hasC) {
            sheet.getRange(row + i, 4).clearContent();
            sheet.getRange(row + i, 12).clearContent();
          }
          continue;
        }

        var finalName = '';
        var foundStatus = '';
        var foundPrice = '';

        // 뷰어 탭에서 코드 검색
        for (var v = 0; v < vData.length; v++) {
          var vCode = cleanCode_(vData[v][2]);
          if (vCode.indexOf('#REF') !== -1 || vCode.indexOf('#N/A') !== -1) continue;
          if (vCode === inputCode) {
            finalName = vData[v][3];   // D열 = 품목명
            foundStatus = vData[v][0]; // A열 = 상태
            foundPrice = vData[v][6];  // G열 = 최종단가
            break;
          }
        }

        // ★ D열(품목명) + L열(단가) 값 기록
        if (finalName) {
          sheet.getRange(row + i, 4).setValue(finalName);
          sheet.getRange(row + i, 12).setValue(foundPrice || '');
        } else {
          sheet.getRange(row + i, 4).setValue('');
          sheet.getRange(row + i, 12).setValue('');
        }

        // ★ D/L열 직접 편집 시 안내 (C열 편집이 아닌 경우만)
        if (!hasC && (hasD || hasL)) {
          try { ss.toast('단가/품목명은 자동입력 열입니다.\n원래 데이터로 복구되었습니다.', '⚠️ 자동 복구', 4); } catch(eToast){}
        }

        // J열(적요) 경고 처리
        var jCell = sheet.getRange(row + i, 10);
        var jNow = String(jCell.getValue() || '').trim();
        if (inputCode && !finalName) {
          // 코드오류
          if (jNow.indexOf('코드오류') === -1) {
            var debugInfo = '코드오류 (총 ' + vData.length + '행 중 일치없음 / 입력: ' + inputCode + ')';
            jCell.setValue(String.fromCharCode(0xD83D, 0xDEA8) + debugInfo);
          }
        } else if (foundStatus && (String(foundStatus).indexOf('품절') !== -1 || String(foundStatus).indexOf('단종') !== -1 || String(foundStatus).indexOf('재고까지만') !== -1)) {
          var warn = String.fromCharCode(0xD83D, 0xDEA8) + ' ' + foundStatus;
          if (jNow !== warn) jCell.setValue(warn);
        } else if (jNow.indexOf('코드오류') !== -1 || jNow.indexOf(String.fromCharCode(0xD83D, 0xDEA8)) === 0) {
          // 매칭 성공 → 기존 경고 삭제
          jCell.clearContent();
        }
      }
    }

    // ★ 필수필드 완성도 감지: N열(상태) 자동 갱신
    if (hasC || hasEH) {
      try {
        var nCol = 14;
        for (var ci = 0; ci < numRows; ci++) {
          var cr = row + ci;
          var cV = String(sheet.getRange(cr, 3).getValue() || '').trim();
          if (!cV) continue;
          var eV = String(sheet.getRange(cr, 5).getValue() || '').trim();
          var fV = String(sheet.getRange(cr, 6).getValue() || '').trim();
          var gV = String(sheet.getRange(cr, 7).getValue() || '').trim();
          var hV = String(sheet.getRange(cr, 8).getValue() || '').trim();
          var miss = [];
          if (!eV || eV === '0') miss.push('수량');
          if (!fV) miss.push('수취인');
          if (!gV) miss.push('전화번호');
          if (!hV) miss.push('주소');
          var nCell = sheet.getRange(cr, nCol);
          var nNow = String(nCell.getValue() || '').trim();
          if (miss.length > 0) {
            var wm = '⚠️입력미완(' + miss.join(',') + ')';
            if (nNow !== wm) nCell.setValue(wm);
          } else if (nNow.indexOf('입력미완') !== -1) {
            nCell.clearContent();
          }
        }
      } catch (eN) {}
    }
  } catch (err) {}
}
